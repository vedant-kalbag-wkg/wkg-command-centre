```typescript
import { Pool } from "pg";
import { readFileSync } from "node:fs";

interface InvariantResult {
  name: string;
  status: "pass" | "fail" | "warn";
  expected?: number | string;
  actual?: number | string;
  detail?: string;
}

const CHECK_FILTER = process.argv.find(a => a.startsWith("--check="))?.split("=")[1] ?? null;
const REPORT_OUT = process.argv.find(a => a.startsWith("--out="))?.split("=")[1] ?? null;

// Parse golden constants from Plan A report
function loadGoldens(): Record<string, number> {
  const reportPath = ".planning/phases/07-data-foundation-rebuild/07-PREFLIGHT-REPORT.md";
  const raw = readFileSync(reportPath, "utf8");
  const goldens: Record<string, number> = {};
  for (const m of raw.matchAll(/export const (GOLDEN_[A-Z_]+) = (\d+);/g)) {
    goldens[m[1]] = parseInt(m[2], 10);
  }
  return goldens;
}

async function runInvariants(client: any, goldens: Record<string, number>): Promise<InvariantResult[]> {
  const results: InvariantResult[] = [];

  const include = (family: string) => CHECK_FILTER === null || CHECK_FILTER === family;

  if (include("locations")) {
    const r = await client.query(`SELECT COUNT(*)::int AS c FROM locations WHERE archived_at IS NULL`);
    const actual = r.rows[0].c;
    results.push({
      name: "locations.active count vs golden",
      status: actual === goldens.GOLDEN_LOCATIONS_ACTIVE ? "pass" : "fail",
      expected: goldens.GOLDEN_LOCATIONS_ACTIVE,
      actual,
    });
  }

  if (include("kiosks")) {
    const r = await client.query(`SELECT COUNT(*)::int AS c FROM kiosks WHERE archived_at IS NULL`);
    const actual = r.rows[0].c;
    results.push({
      name: "kiosks.active count vs golden",
      status: actual === goldens.GOLDEN_KIOSKS_ACTIVE ? "pass" : "fail",
      expected: goldens.GOLDEN_KIOSKS_ACTIVE,
      actual,
    });
  }

  if (include("sales")) {
    const r = await client.query(`SELECT COUNT(*)::bigint AS c FROM sales_records`);
    const actual = parseInt(r.rows[0].c, 10);
    results.push({
      name: "sales_records count vs golden",
      status: actual === goldens.GOLDEN_SALES_RECORDS ? "pass" : "fail",
      expected: goldens.GOLDEN_SALES_RECORDS,
      actual,
    });
    // sales_records gross revenue: net_amount + vat_amount are numeric(12,2)
    // pounds per src/db/schema.ts:678. Compare as the exact decimal string
    // (cast to text) — avoids float-precision drift on a numeric corpus.
    const sum = await client.query(`SELECT COALESCE(SUM(net_amount + vat_amount), 0)::text AS s FROM sales_records`);
    const actualSum: string = sum.rows[0].s;
    results.push({
      name: "sales_records total revenue (gross GBP) vs golden",
      status: actualSum === goldens.GOLDEN_TOTAL_REVENUE_GROSS_GBP ? "pass" : "fail",
      expected: goldens.GOLDEN_TOTAL_REVENUE_GROSS_GBP,
      actual: actualSum,
    });
  }

  if (include("orphans")) {
    // No orphan kiosk_assignments — every assignment must point at a live kiosk + live location
    const r = await client.query(`
      SELECT COUNT(*)::int AS c
      FROM kiosk_assignments ka
      LEFT JOIN kiosks k ON k.id = ka.kiosk_id AND k.archived_at IS NULL
      LEFT JOIN locations l ON l.id = ka.location_id AND l.archived_at IS NULL
      WHERE k.id IS NULL OR l.id IS NULL
    `);
    const actual = r.rows[0].c;
    results.push({
      name: "no orphan kiosk_assignments (FK to live kiosk + live location)",
      status: actual === 0 ? "pass" : "fail",
      expected: 0,
      actual,
    });
  }

  if (include("same_name")) {
    // Same-name invariant via Plan D's helper — but called via raw SQL here to avoid
    // pulling Drizzle into a pg-Pool script. Mirror the helper's query exactly.
    const r = await client.query(`
      SELECT COUNT(*)::int AS c FROM (
        SELECT normalised_name FROM locations
        WHERE archived_at IS NULL AND normalised_name IS NOT NULL
          AND normalised_name != 'locationneeded'
        GROUP BY normalised_name HAVING COUNT(*) > 1
      ) g
    `);
    const actual = r.rows[0].c;
    results.push({
      name: "no active same-name groups (excluding sentinel)",
      status: actual === 0 ? "pass" : "fail",
      expected: 0,
      actual,
    });
  }

  if (include("sentinel")) {
    // LOCATION_NEEDED sentinel: orphan count is surfaced as WARN, not FAIL — Test 3
    const sentinel = await client.query(`
      SELECT id FROM locations WHERE outlet_code = '__LOCATION_NEEDED__' AND name = 'LOCATION_NEEDED' LIMIT 1
    `);
    if (sentinel.rows.length === 0) {
      results.push({
        name: "LOCATION_NEEDED sentinel exists",
        status: "fail",
        expected: "1 row",
        actual: "0 rows",
      });
    } else {
      results.push({ name: "LOCATION_NEEDED sentinel exists", status: "pass", expected: "1 row", actual: "1 row" });
      const orphans = await client.query(`
        SELECT COUNT(DISTINCT kiosk_id)::int AS c FROM kiosk_assignments WHERE location_id = $1
      `, [sentinel.rows[0].id]);
      results.push({
        name: "LOCATION_NEEDED orphan kiosk count (informational)",
        status: "warn",
        actual: orphans.rows[0].c,
        detail: orphans.rows[0].c === 0 ? "no orphans — clean" : `${orphans.rows[0].c} orphan kiosks pending triage`,
      });
    }
  }

  if (include("assigned_at")) {
    // Two-pass backfill coverage: NULL count among non-archived assignments
    const r = await client.query(`
      SELECT COUNT(*)::int AS c FROM kiosk_assignments WHERE assigned_at IS NULL
    `);
    const actual = r.rows[0].c;
    results.push({
      name: "kiosk_assignments.assigned_at coverage (NULL count after two-pass backfill)",
      status: actual === 0 ? "pass" : "warn",  // warn: backfill still incomplete (e.g. kiosks with no live_date and no sales)
      expected: 0,
      actual,
      detail: actual === 0 ? undefined : `${actual} assignments still NULL — likely kiosks with no live_date and no sales corpus coverage`,
    });
  }

  if (include("audit_log")) {
    const r = await client.query(`
      SELECT COUNT(*)::int AS c FROM audit_logs
      WHERE action = 'reseed' AND actor_id = '00000000-0000-0000-0000-000000000001'
    `);
    const actual = r.rows[0].c;
    results.push({
      name: "audit_logs has reseed entry from runbook system actor",
      status: actual >= 1 ? "pass" : "fail",
      expected: ">= 1",
      actual,
    });
  }

  return results;
}

function renderMarkdown(results: InvariantResult[]): string {
  const lines: string[] = ["# Verify Data Reset Report", ""];
  const summary = { pass: 0, fail: 0, warn: 0 };
  for (const r of results) summary[r.status]++;
  lines.push(`**Total:** ${results.length} | **Pass:** ${summary.pass} | **Fail:** ${summary.fail} | **Warn:** ${summary.warn}`);
  lines.push("");
  lines.push("| Invariant | Status | Expected | Actual | Detail |");
  lines.push("|-----------|--------|----------|--------|--------|");
  for (const r of results) {
    lines.push(`| ${r.name} | ${r.status.toUpperCase()} | ${r.expected ?? "—"} | ${r.actual ?? "—"} | ${r.detail ?? ""} |`);
  }
  return lines.join("\n") + "\n";
}

async function main() {
  const url = process.env.DATABASE_URL;
  if (!url) throw new Error("DATABASE_URL not set");
  const pool = new Pool({ connectionString: url });
  const client = await pool.connect();
  try {
    const goldens = loadGoldens();
    const results = await runInvariants(client, goldens);
    const payload = { timestamp: new Date().toISOString(), results, summary: {
      total: results.length,
      pass: results.filter(r => r.status === "pass").length,
      fail: results.filter(r => r.status === "fail").length,
      warn: results.filter(r => r.status === "warn").length,
    } };
    const json = JSON.stringify(payload, null, 2);
    process.stdout.write(json + "\n");
    process.stderr.write(renderMarkdown(results));
    if (REPORT_OUT) {
      const fs = await import("node:fs");
      fs.writeFileSync(REPORT_OUT, json);
    }
    process.exit(payload.summary.fail > 0 ? 1 : 0);
  } finally {
    client.release();
    await pool.end();
  }
}
main().catch((e) => { console.error(e); process.exit(1); });
```
