   export type HotelLocationImportResult = {

     locationsSkipped: number;
     boardsProcessed: number;
     durationMs: number;

   export type HotelLocationImportDeps = {
     mondayApiToken: string;

     resolveRegionId: (boardId: number) => Promise<string>;

   export async function runHotelLocationImport(
     deps: HotelLocationImportDeps,
   ): Promise<HotelLocationImportResult> {
     const { mondayApiToken: _t, db, resolveRegionId, logger = noopLogger } = deps;
     const t0 = Date.now();
     let locationsUpserted = 0;
     let locationsSkipped = 0;

     for (const boardId of HOTEL_BOARD_IDS) {
       logger("hotel-import", `board=${boardId} start`);
       const regionId = await resolveRegionId(boardId);

       for await (const item of iterateBoardItems(boardId)) {
         // Map Monday columns: outlet_code from mirror9, address from text/location columns
         const cols = mapColumnValues(item, {
           outletCode: "mirror9",
           address: "location_mkmqsv5h", // verify exact column id in import-location-products.ts
         });
         const row = {
           name: item.name,
           normalisedName: normaliseName(item.name),
           outletCode: cols.outletCode?.text?.trim() ?? `MONDAY-${item.id}`,
           address: cols.address?.text ?? null,
           primaryRegionId: regionId,
         };

         // Idempotent upsert: dedup by (name, primaryRegionId) — Monday item id is the natural key
         // but locations.id is uuid defaultRandom, so use ON CONFLICT against a unique constraint
         // that already exists (verify in src/db/schema.ts — likely (primary_region_id, outlet_code))
         const result = await db.insert(locations).values(row)
           .onConflictDoNothing({ target: [locations.primaryRegionId, locations.outletCode] })
           .returning({ id: locations.id });
         if (result.length > 0) locationsUpserted++; else locationsSkipped++;
       }
       logger("hotel-import", `board=${boardId} done`);
     }

     return {
       locationsUpserted,
       locationsSkipped,
       boardsProcessed: HOTEL_BOARD_IDS.length,
       durationMs: Date.now() - t0,
     };
   }
