**Step 1 — REPLACE `src/app/(app)/locations/merge-action.ts`** entirely (current is a 12-line thin wrapper). New file (per PATTERNS.md):

```typescript
"use server";
import { requireRole } from "@/lib/auth";
import { db } from "@/db/client";
import { sql } from "drizzle-orm";
import { applyLocationMerge } from "@/lib/location-merge";
import { revalidateTag } from "next/cache";

const LOCK_KEY = 738294108; // location merge UI — distinct from 738294105/106/107

export type MergeLocationsResult =
  | { success: true; merged: number }
  | { error: string }
  | { status: "lock_contention" };

export async function mergeLocationsAction(
  canonicalId: string,
  defunctIds: string[],
): Promise<MergeLocationsResult> {
  const session = await requireRole("admin");

  const lockResult = await db.execute(sql`SELECT pg_try_advisory_lock(${LOCK_KEY})::boolean AS lock`);
  const lockRows = lockResult && "rows" in lockResult
    ? (lockResult as { rows: Array<{ lock: boolean }> }).rows
    : (lockResult as unknown as Array<{ lock: boolean }>);
  const acquired = lockRows[0]?.lock === true;
  if (!acquired) return { status: "lock_contention" };

  try {
    const actor = { id: session.user.id, name: session.user.name ?? session.user.email };
    const result = await applyLocationMerge(canonicalId, defunctIds, actor, db);
    revalidateTag("locations");
    return { success: true, merged: result.locationsArchived };
  } catch (err) {
    return { error: err instanceof Error ? err.message : "Failed to merge locations" };
  } finally {
    try { await db.execute(sql`SELECT pg_advisory_unlock(${LOCK_KEY})`); } catch { /* ignore */ }
  }
```

Function signature is fixed per checker direction:

```typescript
export async function undoMerge(snapshotId: string): Promise<{ success: true } | { error: string }>
```

The action takes a `snapshotId` (the `location_merge_snapshots.id` UUID), NOT the merge audit-log id. The detail page derives the snapshot id from the audit-log row at render-time and passes it to the button (which then passes it here on click).

Pattern: copy the auth + advisory-lock + transaction shape from `src/lib/multi-pos-merge.ts` (cited in PATTERNS.md). Use `pg_advisory_xact_lock(738294108)` — the SAME numeric key as the forward-merge UI but the transaction-scoped variant. Rationale: this intentionally serialises an in-flight forward-merge against any concurrent undo on potentially-overlapping rows; `pg_advisory_xact_lock` auto-releases at COMMIT/ROLLBACK so no manual unlock is needed.

```typescript
"use server";
import { requireRole } from "@/lib/auth";
import { db } from "@/db/client";
import { sql, eq, inArray } from "drizzle-orm";
import {
  locationMergeSnapshots,
  locations,
  kioskAssignments,
  salesRecords,
  // ... import every FK table named in the snapshot.fk_changes payload key set
} from "@/db/schema";
import { writeAuditLog } from "@/lib/audit";
import { revalidateTag } from "next/cache";

export async function undoMerge(
  snapshotId: string,
): Promise<{ success: true } | { error: string }> {
  const session = await requireRole("admin");
  const actor = { id: session.user.id, name: session.user.name ?? session.user.email };

  try {
    return await db.transaction(async (tx) => {
      // (a) Acquire transaction-scoped advisory lock on the SAME key as forward-merge
      //     so a concurrent forward-merge in flight blocks this undo.
      //     Auto-released at COMMIT/ROLLBACK — no finally unlock needed.
      await tx.execute(sql`SELECT pg_advisory_xact_lock(738294108)`);

      // (b) SELECT ... FOR UPDATE the snapshot row by id.
      //     Abort with `error: 'snapshot_already_undone'` if missing
      //     (snapshot deleted = either already undone OR aged out by 30-day cron).
      const snap = await tx.execute(sql`
        SELECT id, audit_log_id, payload, created_at
        FROM ${locationMergeSnapshots}
        WHERE id = ${snapshotId}
        FOR UPDATE
      `);
      const snapRow = snap.rows?.[0] ?? (snap as unknown as Array<any>)[0];
      if (!snapRow) {
        return { error: "snapshot_already_undone" };
      }

      const payload = snapRow.payload as {
        archived_ids: string[];
        // fk_changes: each entry records the prior FK value per row, keyed by table:
        //   { table: "kiosk_assignments", row_id: "<uuid>", fk_column: "location_id", previous_value: "<uuid>" }
        fk_changes: Array<{
          table: string;
          row_id: string;
          fk_column: string;
          previous_value: string;
        }>;
      };

      // (c) Reverse every FK migration recorded in snapshot.fk_changes.
      //     The snapshot stores the prior FK value per row — replay restores it directly.
      //     Use a switch over `table` to dispatch to the correct typed UPDATE,
      //     OR (preferred for safety against table-name injection) a hardcoded allowlist
      //     of {table → drizzle-table object} that throws on any unknown table name.
      const TABLE_DISPATCH: Record<string, any> = {
        kiosk_assignments: kioskAssignments,
        sales_records: salesRecords,
        // ... add every table name appearing in fk_changes
      };
      for (const change of payload.fk_changes) {
        const target = TABLE_DISPATCH[change.table];
        if (!target) {
          throw new Error(`undoMerge: unknown table in fk_changes: ${change.table}`);
        }
        // Build the UPDATE via raw sql for the dynamic column name (Drizzle does not
        // type-safely allow dynamic columns; sql.identifier guards the column name).
        await tx.execute(sql`
          UPDATE ${target}
          SET ${sql.identifier(change.fk_column)} = ${change.previous_value}
          WHERE id = ${change.row_id}
        `);
      }

      // (d) Restore archived_at = NULL on the N-1 archived rows.
      if (payload.archived_ids.length > 0) {
        await tx.update(locations)
          .set({ archivedAt: null })
          .where(inArray(locations.id, payload.archived_ids));
      }

      // (e) Insert an audit_log row of action `location_merge_undone` citing actor + snapshotId + restored ids.
      await writeAuditLog({
        actorId: actor.id,
        actorName: actor.name,
        entityType: "location",
        entityId: payload.archived_ids[0] ?? snapshotId,
        action: "location_merge_undone",
        metadata: {
          snapshotId,
          undidAuditLogId: snapRow.audit_log_id,
          restoredLocationIds: payload.archived_ids,
          fkChangesReversed: payload.fk_changes.length,
        },
      }, tx);

      // (f) DELETE FROM location_merge_snapshots WHERE id = $1.
      //     This is the single source of truth for "undo no longer available".
      //     The detail page checks for snapshot existence to render the button.
      await tx.execute(sql`DELETE FROM location_merge_snapshots WHERE id = ${snapshotId}`);

      revalidateTag("locations");
      return { success: true };
    });
  } catch (err) {
    return { error: err instanceof Error ? err.message : "Failed to undo merge" };
  }
}
```

**Snapshot payload contract.** This action requires the snapshot's JSONB payload to have the shape `{ archived_ids: string[], fk_changes: Array<{ table, row_id, fk_column, previous_value }> }`. Update the forward-merge `applyLocationMerge` in `src/lib/location-merge.ts` (Task 1) to write this exact shape into `location_merge_snapshots.payload` — i.e. for every FK table the merge rewrites, capture per-row `{ table, row_id, fk_column, previous_value }` BEFORE the UPDATE so the snapshot can replay it. The `archived_ids` field is the `defunctIds` array (the rows whose `archived_at` is being set during the merge).
