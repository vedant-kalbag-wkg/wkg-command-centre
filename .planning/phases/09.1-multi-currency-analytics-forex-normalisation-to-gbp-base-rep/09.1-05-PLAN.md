<objective>
Two interlocking surfaces:
1. **Live ETL stamping** — every new `sales_records` row written via the CSV pipeline (admin path + Azure ETL inherits via the same pipeline) gets `net_amount_gbp` populated at insert. Stale rates (> 7d) hard-fail the blob with an alert.
2. **Historical backfill** — every existing row's `net_amount_gbp` is populated via a one-shot script. Once `SELECT COUNT(*) WHERE net_amount_gbp IS NULL = 0`, operator applies migration 0048 to flip the column to NOT NULL.

This plan is `autonomous: false` because it contains an operator-gated migration step (Pitfall 7).

    ```typescript
     * Phase 9.1 Plan 09.1-05 — backfill sales_records.net_amount_gbp historical (FX-02).
     *
     * Run: npx tsx --env-file=.env.neon-dev --tsconfig tsconfig.json scripts/backfill-net-amount-gbp.ts [--dry-run]
     *
     * Idempotent: WHERE net_amount_gbp IS NULL — re-running on already-stamped
     * rows is a no-op (D-04). Restartable: cursor-based on (transaction_date, id).
     * GBP rows: identity shortcut (no rate lookup). Others: carry-forward from
     * exchange_rates (D-05). Hard-fails if no rate exists OR staleDays > 7 (D-07).
     */
    import { Pool } from "pg";

    const DRY_RUN = process.argv.includes("--dry-run");
    const CHUNK = 1000;

    type Row = {
      id: string;
      currency: string;
      net_amount: string;
      transaction_date: string;
    };

    async function lookupRate(
      client: Awaited<ReturnType<Pool["connect"]>>,
      currency: string,
      isoDate: string,
    ): Promise<{ rate: number; rateDate: string; staleDays: number } | null> {
      if (currency === "GBP") return { rate: 1.0, rateDate: isoDate, staleDays: 0 };
      const { rows } = await client.query<{ rate_to_gbp: string; rate_date: string }>(
        `SELECT rate_to_gbp, rate_date::text FROM exchange_rates
         WHERE currency = $1 AND rate_date <= $2
         ORDER BY rate_date DESC LIMIT 1`,
        [currency, isoDate],
      );
      if (rows.length === 0) return null;
      const r = rows[0];
      const staleDays = Math.floor(
        (Date.parse(isoDate) - Date.parse(r.rate_date)) / 86400000,
      );
      return { rate: Number(r.rate_to_gbp), rateDate: r.rate_date, staleDays };
    }

    async function main() {
      const url = process.env.DATABASE_URL;
      if (!url) throw new Error("DATABASE_URL not set");
      console.log("Target:", url.replace(/:[^:@]+@/, ":***@"));
      console.log("Mode:  ", DRY_RUN ? "DRY RUN (no writes)" : "WRITE");

      const pool = new Pool({ connectionString: url });
      const client = await pool.connect();
      let totalUpdated = 0;
      try {
        await client.query("BEGIN");
        let cursorDate: string | null = null;
        let cursorId: string | null = null;
        // Cache rates within a single backfill run for performance
        const rateCache = new Map<string, { rate: number; staleDays: number }>();

        while (true) {
          const { rows } = cursorDate && cursorId
            ? await client.query<Row>(
                `SELECT id::text, currency, net_amount::text, transaction_date::text
                 FROM sales_records
                 WHERE net_amount_gbp IS NULL
                   AND (transaction_date, id::text) > ($1, $2)
                 ORDER BY transaction_date, id::text
                 LIMIT $3`,
                [cursorDate, cursorId, CHUNK],
              )
            : await client.query<Row>(
                `SELECT id::text, currency, net_amount::text, transaction_date::text
                 FROM sales_records
                 WHERE net_amount_gbp IS NULL
                 ORDER BY transaction_date, id::text
                 LIMIT $1`,
                [CHUNK],
              );

          if (rows.length === 0) break;

          const ids: string[] = [];
          const stamps: string[] = [];
          for (const row of rows) {
            const key = `${row.currency}|${row.transaction_date}`;
            let r = rateCache.get(key);
            if (!r) {
              const looked = await lookupRate(client, row.currency, row.transaction_date);
              if (!looked) {
                throw new Error(
                  `Backfill halt: no FX rate exists for ${row.currency} on or before ${row.transaction_date} (first affected row id ${row.id})`,
                );
              }
              if (looked.staleDays > 7) {
                throw new Error(
                  `Backfill halt: stale FX rate for ${row.currency} on ${row.transaction_date} (staleDays=${looked.staleDays} > 7; first affected row id ${row.id})`,
                );
              }
              r = { rate: looked.rate, staleDays: looked.staleDays };
              rateCache.set(key, r);
            }
            const stamp =
              row.currency === "GBP"
                ? row.net_amount
                : (Number(row.net_amount) * r.rate).toFixed(2);
            ids.push(row.id);
            stamps.push(stamp);
          }

          await client.query(
            `UPDATE sales_records SET net_amount_gbp = data.gbp::numeric
             FROM (
               SELECT unnest($1::uuid[]) AS id,
                      unnest($2::numeric[]) AS gbp
             ) AS data
             WHERE sales_records.id = data.id`,
            [ids, stamps],
          );

          totalUpdated += rows.length;
          cursorDate = rows[rows.length - 1].transaction_date;
          cursorId = rows[rows.length - 1].id;
          console.log(`Progress: ${totalUpdated} rows stamped (last ${cursorDate} / ${cursorId})`);
        }

        if (DRY_RUN) {
          await client.query("ROLLBACK");
          console.log(`\nDry run — rolled back. Would have stamped ${totalUpdated} rows.`);
        } else {
          await client.query("COMMIT");
          console.log(`\nCommitted. Stamped ${totalUpdated} rows.`);
        }
      } catch (err) {
        await client.query("ROLLBACK");
        console.error("\nRolled back:", err);
        throw err;
      } finally {
        client.release();
        await pool.end();
      }
    }

    main().catch((e) => {
      console.error(e);
      process.exit(1);
    });
    ```
