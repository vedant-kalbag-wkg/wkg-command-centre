```typescript
export async function getRateForDate(
  currency: string,
  date: string,  // ISO 'YYYY-MM-DD'
): Promise<{ rate: number; rateDate: string; staleDays: number } | null> {
  if (currency === 'GBP') return { rate: 1.0, rateDate: date, staleDays: 0 };
  const [row] = await db
    .select({ rate: exchangeRates.rateToGbp, rateDate: exchangeRates.rateDate })
    .from(exchangeRates)
    .where(and(eq(exchangeRates.currency, currency), lte(exchangeRates.rateDate, date)))
    .orderBy(desc(exchangeRates.rateDate))
    .limit(1);
  if (!row) return null;
  const staleDays = Math.floor((Date.parse(date) - Date.parse(row.rateDate)) / 86400000);
  return { rate: Number(row.rate), rateDate: row.rateDate, staleDays };
}
```
