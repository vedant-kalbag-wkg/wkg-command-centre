    ```typescript
    import { z } from "zod";
    import { BOE_SERIES_TO_CCY } from "./currencies";

    export type ParsedRate = { currency: string; rateDate: string; rate: number };

    const ParsedRateSchema = z.object({
      currency: z.string().refine(c => Object.values(BOE_SERIES_TO_CCY).includes(c), {
        message: "currency not in BOE_SERIES_TO_CCY",
      }),
      rateDate: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, "rateDate must be ISO YYYY-MM-DD"),
      rate: z.number().positive().finite(),
    });

    const MONTHS: Record<string, string> = {
      Jan: "01", Feb: "02", Mar: "03", Apr: "04", May: "05", Jun: "06",
      Jul: "07", Aug: "08", Sep: "09", Oct: "10", Nov: "11", Dec: "12",
    };

    function boeDateToIso(boe: string): string {
      // "08 May 2026" → "2026-05-08"
      const [d, m, y] = boe.trim().split(/\s+/);
      const mm = MONTHS[m];
      if (!mm) throw new Error(`unrecognised BoE month token "${m}" in date "${boe}"`);
      return `${y}-${mm}-${d.padStart(2, "0")}`;
    }

    export function parseBoeCsv(csv: string): ParsedRate[] {
      const lines = csv.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0);
      const headerIdx = lines.findIndex(l => l.toLowerCase().startsWith("date,"));
      if (headerIdx === -1) throw new Error("BoE CSV missing header row starting with 'Date,'");
      const header = lines[headerIdx].split(",").map(c => c.trim());
      const seriesCols = header.slice(1); // skip "Date"
      const out: ParsedRate[] = [];
      for (const line of lines.slice(headerIdx + 1)) {
        const cells = line.split(",").map(c => c.trim());
        const dateCell = cells[0];
        const isoDate = boeDateToIso(dateCell);
        for (let i = 0; i < seriesCols.length; i++) {
          const series = seriesCols[i];
          const ccy = BOE_SERIES_TO_CCY[series as keyof typeof BOE_SERIES_TO_CCY];
          if (!ccy) continue; // unknown series code → skip silently
          const cell = cells[i + 1];
          if (!cell || cell === "") continue; // BoE non-publish (weekend/holiday) → skip
          const rate = Number(cell);
          const parsed = ParsedRateSchema.parse({ currency: ccy, rateDate: isoDate, rate });
          out.push(parsed);
        }
      }
      return out;
    }

    export function buildBoeCsvUrl(
      from: string, // ISO YYYY-MM-DD
      to: string,
      seriesCodes: readonly string[],
    ): string {
      const fmt = (iso: string) => {
        const [y, m, d] = iso.split("-");
        const monthName = Object.entries(MONTHS).find(([_, num]) => num === m)?.[0];
        return `${Number(d)}/${monthName}/${y}`;
      };
      return (
        `https://www.bankofengland.co.uk/boeapps/database/_iadb-FromShowColumns.asp` +
        `?CodeVer=new&csv.x=yes&CSVF=TN&UsingCodes=Y&VPD=Y&VFD=N` +
        `&Datefrom=${fmt(from)}&Dateto=${fmt(to)}` +
        `&SeriesCodes=${seriesCodes.join(",")}`
      );
    }

    export async function fetchBoeRatesForDate(isoDate: string): Promise<ParsedRate[]> {
      const url = buildBoeCsvUrl(isoDate, isoDate, Object.keys(BOE_SERIES_TO_CCY));
      const resp = await fetch(url);
      if (!resp.ok) throw new Error(`BoE fetch HTTP ${resp.status} for ${isoDate}`);
      const csv = await resp.text();
      return parseBoeCsv(csv);
    }
    ```

    ```typescript
    import { and, desc, eq, lte } from "drizzle-orm";
    import { db } from "@/db";
    import { exchangeRates } from "@/db/schema";

    export type RateLookupResult = {
      rate: number;
      rateDate: string;
      staleDays: number;
    };

    /**
     * Look up the GBP rate for `currency` on `isoDate`, with carry-forward.
     *
     * - GBP rows: identity ({ rate: 1.0, ..., staleDays: 0 }) WITHOUT DB call (D-04).
     * - Non-GBP: most-recent rate ≤ isoDate per CONTEXT D-05.
     * - Returns null when no rate exists for the currency on or before isoDate.
     *   Caller (ETL) MUST hard-fail per D-03 — no silent GBP fallback.
     * - staleDays > 7 means the caller MUST hard-fail per D-07 (the lookup
     *   itself does NOT throw on stale; ETL controls the policy).
     */
    export async function getRateForDate(
      currency: string,
      isoDate: string,
    ): Promise<RateLookupResult | null> {
      if (currency === "GBP") {
        return { rate: 1.0, rateDate: isoDate, staleDays: 0 };
      }
      const [row] = await db
        .select({
          rate: exchangeRates.rateToGbp,
          rateDate: exchangeRates.rateDate,
        })
        .from(exchangeRates)
        .where(
          and(eq(exchangeRates.currency, currency), lte(exchangeRates.rateDate, isoDate)),
        )
        .orderBy(desc(exchangeRates.rateDate))
        .limit(1);
      if (!row) return null;
      const staleDays = Math.floor(
        (Date.parse(isoDate) - Date.parse(row.rateDate)) / 86400000,
      );
      return {
        rate: Number(row.rate),
        rateDate: row.rateDate,
        staleDays,
      };
    }
    ```
